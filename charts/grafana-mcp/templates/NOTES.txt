    export NODE_PORT=$(kubectl get --namespace {{ include "grafana-mcp.namespace" . }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ $fullName }})
    export NODE_IP=$(kubectl get nodes --namespace {{ include "grafana-mcp.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
    echo "MCP URL: http://$NODE_IP:$NODE_PORT/mcp"
{{- else if contains "LoadBalancer" .Values.service.type }}

    kubectl get svc --namespace {{ include "grafana-mcp.namespace" . }} {{ $fullName }} -w
{{- else if contains "ClusterIP" .Values.service.type }}
  Forward port to localhost:
    kubectl port-forward --namespace {{ include "grafana-mcp.namespace" . }} svc/{{ $fullName }} 8080:{{ $svcPort }}
    echo "MCP URL: http://localhost:8080/mcp"
